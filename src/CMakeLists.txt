# ==============================================================================
# The layers, as libraries.
#
#   dew_model   the project: ValueTree, schema, edits, serialization. A leaf.
#   dew_engine  the audio: snapshot, transport, sequencer, voices, effects, mix.
#               Opens no files and no devices - that is what makes the whole
#               test suite able to drive it with neither.
#   dew_io      everything that touches the OS: audio and MIDI devices, the
#               sample cache, the offline renderer, the exporters.
#   dew_design  the design system: tokens, icons, primitives, look and feel.
#   dew_app     validated window/view/device state. A leaf that the UI reads.
#   dew_ui      the editors.
#
# They were one static library called dew_core, whose own header comment said
# "model + engine" and which then listed thirty UI translation units. The
# include graph was already acyclic and correctly directed; nothing enforced it,
# and dew_render - which is headless and includes no UI at all - linked every
# piece of the interface and all of juce_gui_extra.
#
# Split, the layering claims are link errors rather than good intentions.
# ==============================================================================

# ------------------------------------------------------------------------------
# Shared JUCE configuration. INTERFACE, so every layer gets the same definitions
# and the same include root without one of them owning the others' settings.
# ------------------------------------------------------------------------------
add_library(dew_juce_config INTERFACE)

target_include_directories(dew_juce_config INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")

target_compile_definitions(dew_juce_config
  INTERFACE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    # MP3 ENCODING. JUCE can only decode MP3 on its own; encoding shells out to
    # an installed lame binary, so this ships no LAME code and adds no build
    # dependency. Without the binary the format simply reports itself
    # unavailable - see OfflineRenderer::isAvailable.
    JUCE_USE_LAME_AUDIO_FORMAT=1
    # Set by juce_add_gui_app / juce_add_console_app on THOSE targets only. A
    # static library shared between them never sees it, and JUCE's headers
    # behave differently without it, so it has to be declared here by hand.
#
    # It is also exactly the definition a plugin build must NOT have, which is
    # why it lives on an interface target that a plugin could decline rather
    # than baked into every layer.
    JUCE_STANDALONE_APPLICATION=1
    DEW_VERSION_STRING="${PROJECT_VERSION}${DEW_VERSION_SUFFIX}"
    DEW_JUCE_PIN="${DEW_PIN_JUCE}"
# The product's name, so the one place that spells it is the project() call.
# The About window and getApplicationName both read it through BuildInfo, which
# is where the rest of this binary's provenance already lives.
    DEW_PRODUCT_NAME="${PROJECT_NAME}"
)

# ------------------------------------------------------------------------------
# JUCE itself, compiled once.
#
# A JUCE module is an INTERFACE library carrying its own .cpp/.mm files in
# INTERFACE_SOURCES, so before this every layer, tool and test binary compiled
# them again: 265 compiles of 25 files, and 1.4 GB of byte-identical objects in
# a 3.1 GB tree. dew_<module> is that module built once. The link EDGES below are
# unchanged - dew_model still gets data_structures and graphics and not
# gui_basics - and so is every define reaching dew's own sources.
# ------------------------------------------------------------------------------
include(JuceOnce)
dew_juce_build_once()

# ------------------------------------------------------------------------------
# dew_lang - the score language. Depends on nothing at all, JUCE included.
#
# A compiler that cannot reach into the document is one whose only output is its
# IR, which is what keeps the boundary honest. It also makes the tests instant -
# no ScopedJuceInitialiser, no message thread - lets diagnostics use plain UTF-8
# byte offsets rather than juce::String's character indices, and keeps parser
# iteration off the DAW's build graph.
#
# The juce_recommended_* targets carry compile options only, no include
# directories, so linking them adds no JUCE while keeping this library under the
# same warnings-as-errors gate as everything else. That it stays JUCE-free is
# checked by a source gate, because a header-only include would still compile.
# ------------------------------------------------------------------------------
# dew_i18n - the string catalogue. Depends on juce_core and nothing else.
#
# The link line is the point. A catalogue that cannot reach juce_data_structures
# cannot grow a ValueTree-backed override table, and one that cannot open a file
# has to be compiled in - which is what keeps it in step with the build that
# ships it. Reading a catalogue off disk would also be file I/O two layers below
# dew_io, the only layer allowed to touch a file at all.
#
# It sits below dew_model rather than inside it because dew_design needs it too,
# and dew_design knows nothing about a document. Only dew_model names it: the
# link is PUBLIC and every layer above inherits it, because naming an inherited
# library a second time makes every link warn about a duplicate.
#
# StringIds.h and Catalogs.cpp are GENERATED into the build tree and never
# committed. Under src/ they would be walked by the source gates, and a
# catalogue is longer than the gate on file length should have an opinion about.
# ------------------------------------------------------------------------------
set(DEW_I18N_LOCALES en)

set(DEW_I18N_INPUTS "")
foreach(dew_locale ${DEW_I18N_LOCALES})
  list(APPEND DEW_I18N_INPUTS "${CMAKE_SOURCE_DIR}/resources/i18n/${dew_locale}.json")
endforeach()

set(DEW_I18N_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/i18n")
set(DEW_LANG_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/lang")

# Named upwards so tests/ can read them rather than spelling the same two paths
# a second time. The test that the two PluralTable.h copies agree is the only
# thing that needs them, and it would be a poor test if it were checking a path
# it had guessed.
set(DEW_I18N_GENERATED "${DEW_I18N_GENERATED}" PARENT_SCOPE)
set(DEW_LANG_GENERATED "${DEW_LANG_GENERATED}" PARENT_SCOPE)

# One command, six outputs, two libraries. The `lang.` subtree of the same
# en.json becomes dew_lang's catalogue - see cmake/GenStrings.cmake - and
# PluralTable.h is written into both directories byte for byte identical,
# because neither library may include the other's header. resources/i18n/
# plurals.txt argues that; a test checks the two copies agree.
add_custom_command(
  OUTPUT "${DEW_I18N_GENERATED}/StringIds.h" "${DEW_I18N_GENERATED}/Catalogs.cpp"
         "${DEW_I18N_GENERATED}/PluralTable.h"
         "${DEW_LANG_GENERATED}/MessageIds.h" "${DEW_LANG_GENERATED}/MessageCatalogs.cpp"
         "${DEW_LANG_GENERATED}/PluralTable.h"
  COMMAND "${CMAKE_COMMAND}"
          "-DDEW_I18N_LOCALES=${DEW_I18N_LOCALES}"
          "-DDEW_I18N_DIR=${CMAKE_SOURCE_DIR}/resources/i18n"
          "-DDEW_I18N_OUT=${DEW_I18N_GENERATED}"
          "-DDEW_LANG_OUT=${DEW_LANG_GENERATED}"
          -P "${CMAKE_SOURCE_DIR}/cmake/GenStrings.cmake"
  DEPENDS ${DEW_I18N_INPUTS}
          "${CMAKE_SOURCE_DIR}/resources/i18n/plurals.txt"
          "${CMAKE_SOURCE_DIR}/cmake/GenStrings.cmake"
  COMMENT "dew: generating the string catalogues"
  VERBATIM
)

add_library(dew_i18n STATIC
  i18n/Arg.cpp
  i18n/PluralRules.cpp
  i18n/MessageFormat.cpp
  i18n/Strings.cpp
  "${DEW_I18N_GENERATED}/Catalogs.cpp"

  "${DEW_I18N_GENERATED}/PluralTable.h"
)

# dew_control - the operation table an agent drives dew through. Links dew_io

#
# The link line is the argument, as it is for dew_lang and dew_i18n. A library
# that cannot see dew_design cannot paint, and one that cannot see dew_ui cannot
# reach a Component - so an operation is a function of the document and the

#

# SoundFontProvider.h already give dew_engine for the same problem: the layer
# below needs something the layer above owns.

# A precompiled header, and the reason it is worth one when dew_tests measured
# SLOWER with the same idea in September: back then every target carried JUCE's
# own 25 translation units, a PCH could not be applied to them (a -include'd
# JUCE header makes juce_*.cpp hard-error, and the .mm ones refuse a PCH built
# with Objective-C off), and the exclusions cost more than the header saved.
# dew_ui now compiles 64 files, all of dew's own and all of them GUI, so the
# JUCE headers they share are precompiled once instead of 64 times.
#
# Measured on the same work - 59 files rebuilt after touching Tokens.h, with
# CCACHE_DISABLE=1 - 25.7 s wall / 181 CPU-s became 14.8 s / 101 CPU-s.
#
# EXTERNAL headers only. dew's own must never go in one: a file could then use
# a layer it does not include, and the include graph is what the layering gates
# in tests/SourceGateTests.cpp read.
target_precompile_headers(dew_ui
  PRIVATE
    <juce_gui_basics/juce_gui_basics.h>
    <vector>
    <optional>)

target_link_libraries(dew_ui
  PUBLIC
    dew_design
    dew_app
    dew_control
    dew_juce_gui_extra
  PRIVATE
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

# ------------------------------------------------------------------------------

juce_add_gui_app(dew
  PRODUCT_NAME              "dew"
  COMPANY_NAME              "dew"
  BUNDLE_ID                 "io.dew.app"
  VERSION                   "${PROJECT_VERSION}"
  MICROPHONE_PERMISSION_ENABLED TRUE
  MICROPHONE_PERMISSION_TEXT "dew records audio from the input you choose in Audio Settings."
  NEEDS_MIDI_INPUT          TRUE
  NEEDS_MIDI_OUTPUT         FALSE
  # Rasterised from resources/icon/dew.svg, which gen-theme.mjs writes from the
  # palette, by `dew_shot icon`. Committed rather than built: JUCE needs the file
  # at CONFIGURE time and the tool that makes it is a target in this project.
  ICON_BIG                  "${CMAKE_SOURCE_DIR}/resources/icon/dew.png"
  DOCUMENT_EXTENSIONS       "dew"
)

target_sources(dew PRIVATE
  main.cpp
  DewApplication.cpp
  DewApplicationMenus.cpp
  DewApplicationPrefs.cpp
)

target_link_libraries(dew
  PRIVATE
    dew_ui
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)
